function onSectionInput() {
    const el = document.getElementById('sectionTextarea') || document.getElementById('sectionInput');
    const sec = SECTIONS[currentSection];
    const count = el.value.length;
    const countEl = document.getElementById('charCount');

    }

    sectionData[currentSection] = el.value;
    localStorage.setItem('thesisData', JSON.stringify(sectionData));
    updateProgress();
}

function updateProgress() {
    const total = Object.keys(SECTIONS).length;
    let filled = 0;
    for (let key in SECTIONS) {
        if (sectionData[key] && sectionData[key].trim().length >= SECTIONS[key].minLength) {
            filled++;
        }
    }
    const bar = document.getElementById('progressBar');
    const fill = document.getElementById('progressFill');
    if (filled > 0) {
        bar.classList.add('show');
        fill.style.width = Math.round((filled / total) * 100) + '%';
    } else {
        bar.classList.remove('show');
    }
}

}

// ============================================
// محرك التحليل الذكي
// ============================================
function performAnalysis(filledSections) {
    const results = {
        totalScore: 0,
        sectionScores: {},
        sectionNotes: {},
        generalNotes: [],
        stats: {
            totalWords: 0,
            totalChars: 0,
            sectionsAnalyzed: filledSections.length,
            highIssues: 0,
            mediumIssues: 0,
            lowIssues: 0,
            infoCount: 0
        },
        wordFrequency: {}
    };

    let totalWeighted = 0;
    let totalWeight = 0;

    // أوزان الأقسام حسب الدور
    const weights = getRoleWeights();

    filledSections.forEach(key => {
        const text = sectionData[key].trim();
        const secResult = analyzeSection(key, text);
        results.sectionScores[key] = secResult.score;
        results.sectionNotes[key] = secResult.notes;

        const w = weights[key] || 1;
        totalWeighted += secResult.score * w;
        totalWeight += w;

        // إحصائيات
        results.stats.totalWords += countWords(text);
        results.stats.totalChars += text.length;

        secResult.notes.forEach(n => {
            if (n.severity === 'high') results.stats.highIssues++;
            else if (n.severity === 'medium') results.stats.mediumIssues++;
            else if (n.severity === 'low') results.stats.lowIssues++;
            else results.stats.infoCount++;
        });

        // تكرار الكلمات
        extractWordFrequency(text, results.wordFrequency);
    });

    results.totalScore = totalWeight > 0 ? Math.round(totalWeighted / totalWeight) : 0;

    // ملاحظات عامة
    results.generalNotes = generateGeneralNotes(filledSections, results);

    return results;
}

function getRoleWeights() {
    const base = {
        title: 1.2,
        toc: 0.8,
        abstract: 1.3,
        objectives: 1.4,
        hypotheses: 1.3,
        theory: 1.5,
        methodology: 1.6,
        analysis: 1.5,
        results: 1.4,
        recommendations: 1.0,
        references: 0.9
    };
    // تعديل الأوزان حسب الدور
    if (currentRole === 'supervisor') {
        base.theory = 1.8;
        base.methodology = 1.8;
        base.objectives = 1.6;
    } else if (currentRole === 'internal') {
        base.analysis = 1.8;
        base.results = 1.7;
        base.hypotheses = 1.6;
    } else if (currentRole === 'external') {
        base.title = 1.5;
        base.abstract = 1.5;
        base.theory = 1.7;
        base.references = 1.4;
    } else if (currentRole === 'researcher') {
        base.methodology = 1.8;
        base.analysis = 1.8;
        base.references = 1.3;
    }
    return base;
}

// تحليل كل قسم على حدة
// ============================================
function analyzeSection(key, text) {
    const score = { value: 100, deductions: [] };
    const notes = [];

    switch(key) {
        case 'title': analyzeTitle(text, score, notes); break;
        case 'toc': analyzeTOC(text, score, notes); break;
        case 'abstract': analyzeAbstract(text, score, notes); break;
        case 'objectives': analyzeObjectives(text, score, notes); break;
        case 'hypotheses': analyzeHypotheses(text, score, notes); break;
        case 'theory': analyzeTheory(text, score, notes); break;
        case 'methodology': analyzeMethodology(text, score, notes); break;
        case 'analysis': analyzeAnalysis(text, score, notes); break;
        case 'results': analyzeResults(text, score, notes); break;
        case 'recommendations': analyzeRecommendations(text, score, notes); break;
        case 'references': analyzeReferences(text, score, notes); break;
    }

    // خصومات عامة للنص
    applyGeneralTextChecks(text, score, notes, key);

    score.value = Math.max(0, Math.min(100, score.value));

    return { score: score.value, notes };
}

// --- تحليل العنوان ---
function analyzeTitle(text, score, notes) {
    const words = text.split(/\s+/).filter(w => w.length > 0);
    const cleanText = text.replace(/[^\u0600-\u06FFa-zA-Z0-9\s]/g, '');

    // الطول
    if (words.length < 5) {
        score.deductions.push(20);
        notes.push(makeNote('high', 'العنوان قصير جداً', `العنوان يحتوي على ${words.length} كلمات فقط، والحد الأدنى المناسب هو 8-15 كلمة لضمان الوضوح والشمولية.`, 'أعد صياغة العنوان ليكون أكثر شمولية مع الإبقاء على الدقة.'));
    } else if (words.length > 25) {
        score.deductions.push(10);
        notes.push(makeNote('medium', 'العنوان طويل جداً', `العنوان يحتوي على ${words.length} كلمة، مما قد يضعف تركيز القارئ على الموضوع الرئيسي.`, 'اخصر العنوان مع الحفاظ على العناصر الأساسية: المتغيرات، المجال، والعينة إن وجدت.'));
    }

    // كلمات عامة
    const vagueWords = ['دراسة', 'بحث', 'بحث في', 'دراسة حول', 'نحو', 'في ضوء'];
    let vagueCount = 0;
    vagueWords.forEach(w => {
        if (text.includes(w)) vagueCount++;
    });
    if (vagueCount >= 2 && words.length < 10) {
        score.deductions.push(10);
        notes.push(makeNote('medium', 'العنوان يحتوي على صياغات عامة', 'وجود أكثر من صياغة عامة دون تحديد دقيق يجعل العنوان ضعيفاً.', 'استبدل الصياغات العامة بمصطلحات محددة تعبر عن محتوى الرسالة بدقة.'));
    }

    // فحص وجود متغيرات

    }

    // فحص الحدود المكانية أو الزمنية
    const hasScope = /في|على|بين|من|خلال|\d{4}/.test(text);
    if (!hasScope) {
        score.deductions.push(5);
        notes.push(makeNote('low', 'العنوان لا يحدد نطاق البحث', 'يفضل أن يحدد العنوان المجال أو المكان أو الزمان لإطار البحث.', 'أضف تحديداً للمجال أو المكان مثل: "في جامعة..." أو "في قطاع..."'));
    }

    // علامات الترقيم
    if (/[!?]/.test(text)) {
        score.deductions.push(5);
        notes.push(makeNote('medium', 'وجود علامات ترقيم غير مناسبة في العنوان', 'العنوان الأكاديمي لا يحتوي على علامات تعجب أو استفهام.', 'أزل علامات الترقيم非المناسبة واستخدم صياغة جملة اسمية.'));
    }
}

// --- تحليل الفهرس ---
function analyzeTOC(text, score, notes) {
    const lines = text.split('\n').filter(l => l.trim().length > 0);

    // عدد الأقسام
    if (lines.length < 5) {
        score.deductions.push(25);
        notes.push(makeNote('high', 'الفهرس فقير جداً', `يحتوي على ${lines.length} عناصر فقط، مما يشير إلى هيكل غير مكتمل.`, 'تأكد من إدراج جميع فصول الرسالة وأبوابها الفرعية مع ترقيم الصفحات.'));
    } else if (lines.length < 10) {
        score.deductions.push(12);
        notes.push(makeNote('medium', 'الفهرس قد يكون غير مكتمل', `عدد العناصر ${lines.length} وهو أقل من المتوقع لرسالة أكاديمية متكاملة.`, 'راجع الفهرس وتأكد من تغطية جميع الأقسام الفرعية.'));
    }

    // فحص الأقسام الأساسية
    const requiredSections = ['مقدمة', 'الإطار النظري', 'المنهج', 'النتائج', 'التوصيات', 'المراجع', 'المستخلص'];
    let missing = [];
    requiredSections.forEach(sec => {
        if (!text.includes(sec)) missing.push(sec);
    });
    if (missing.length > 0) {
        score.deductions.push(missing.length * 5);
        notes.push(makeNote('high', 'أقسام أساسية مفقودة من الفهرس', `الأقسام التالية غير موجودة: ${missing.join('، ')}`, 'أضف الأقسام المفقودة مع ترقيم الصفحات المناسبة.'));
    }

    // فحص ترقيم الصفحات
    const hasPageNums = /\d{1,4}\s*$/.test(text) || /\d{1,4}\s*\./.test(text);
    if (!hasPageNums && lines.length > 3) {
        score.deductions.push(10);
        notes.push(makeNote('medium', 'لا توجد أرقام صفحات في الفهرس', 'الفهرس بدون ترقيم صفحات يفقد وظيفته الأساسية.', 'أضف أرقام الصفحات مقابل كل عنوان فرعي.'));
    }

    // فحص التسلسل المنطقي
    const logicalOrder = ['مستخلص', 'مقدمة', 'إطار', 'منهج', 'نتائج', 'توصيات', 'مراجع'];
    let orderOk = true;
    let lastIndex = -1;
    logicalOrder.forEach((term, idx) => {
        const pos = text.indexOf(term);
        if (pos !== -1) {
            if (pos < lastIndex) orderOk = false;
            lastIndex = pos;
        }
    });
    if (!orderOk) {
        score.deductions.push(8);
        notes.push(makeNote('low', 'ترتيب الأقسام قد لا يتبع التسلسل المنطقي المعتاد', 'التسلسل المعتاد: مستخلص، مقدمة، إطار نظري، منهجية، نتائج، توصيات، مراجع.', 'راجع ترتيب الفصول ليتوافق مع التسلسل الأكاديمي المعتمد.'));
    }
}

// --- تحليل المستخلص ---
function analyzeAbstract(text, score, notes) {
    const words = countWords(text);

    // الطول
    if (words < 100) {
        score.deductions.push(20);
        notes.push(makeNote('high', 'المستخلص قصير جداً', `يحتوي على ${words} كلمة فقط، والحد الأدنى المقبول 200 كلمة تقريباً.`, 'وسّع المستخلص ليشمل: المشكلة، الهدف، المنهج، العينة، أهم النتائج.'));
    } else if (words > 800) {
        score.deductions.push(8);
        notes.push(makeNote('low', 'المستخلص أطول من المعتاد', `يحتوي على ${words} كلمة، والمستخلص المثالي 200-500 كلمة.`, 'اختصر المستخلص مع الحفاظ على العناصر الأساسية.'));
    }

    // العناصر المطلوبة
    const elements = [
        { term: ['مشكلة', 'إشكالية', 'تحدّي'], name: 'المشكلة' },
        { term: ['هدف', 'أهداف', 'يسعى', 'يهدف'], name: 'الهدف' },
        { term: ['منهج', 'أسلوب', 'طريقة', 'منهجية', 'وصفي', 'تجريبي', 'تحليلي'], name: 'المنهج' },
        { term: ['عينة', 'فرض', 'مجتمع', 'مبحوث', 'مستجيب', 'أفراد'], name: 'العينة' },
        { term: ['نتائج', 'توصلت', 'أظهرت', 'تبين', 'أشارت'], name: 'النتائج' },
        { term: ['توصية', 'يقترح', 'توصيات'], name: 'التوصيات' }
    ];

    let missingElements = [];
    elements.forEach(el => {
        const found = el.term.some(t => text.includes(t));
        if (!found) missingElements.push(el.name);
    });

    if (missingElements.length >= 3) {
        score.deductions.push(20);
        notes.push(makeNote('high', 'المستخلص يفتقر لعناصر جوهرية', `العناصر المفقودة: ${missingElements.join('، ')}`, 'أعد كتابة المستخلص ليشمل جميع العناصر الأساسية بالتسلسل المنطقي.');
    } else if (missingElements.length > 0) {
        score.deductions.push(missingElements.length * 5);
        notes.push(makeNote('medium', 'المستخلص يفتقر لبعض العناصر', `العناصر المفقودة: ${missingElements.join('، ')}`, 'أضف العناصر المفقودة لجعل المستخلص متكاملاً.'));
    }

    // فحص الأرقام والإحصائيات
    if (!/\d/.test(text)) {
        score.deductions.push(8);
        notes.push(makeNote('low', 'المستخلص لا يحتوي على أية أرقام أو إحصائيات', 'وجود أرقام (حجم العينة، النسب المئوية) يعزز مصداقية المستخلص.', 'أضف بعض الأرقام الأساسية مثل حجم العينة وأهم النسب الإحصائية.'));
    }

    // فحص الكلمات العامية
    checkColloquialWords(text, score, notes);
}

// --- تحليل الأهداف ---
function analyzeObjectives(text, score, notes) {
    const lines = text.split('\n').filter(l => l.trim().length > 5);

    if (lines.length < 2) {
        score.deductions.push(20);
        notes.push(makeNote('high', 'عدد الأهداف غير كافٍ', 'رسالة الماجستير تحتاج 3-5 أهداف على الأقل، والدكتوراه 4-6 أهداف.', 'حدد أهدافاً إضافية تغطي جوانب البحث المختلفة.');
    } else if (lines.length > 8) {
        score.deductions.push(8);
        notes.push(makeNote('medium', 'عدد الأهداف كبير جداً', `يوجد ${lines.length} هدف، مما قد يؤدي إلى تشتت البحث.`, 'دمج بعض الأهداف المتقاربة لتقليل عددها إلى 3-6 أهداف.');
    }

    // فحص صياغة الأهداف (يجب أن تبدأ بأفعال قابلة للقياس)
    const goodStarters = ['تحديد', 'قياس', 'تقييم', 'معرفة', 'كشف', 'تحليل', 'استكشاف', 'دراسة', 'فحص', 'التحقق', ' تقديم', 'وصف', 'معرفة مدى'];
    const badStarters = ['التعرف على', 'الوقوف على', 'إلقاء الضوء', 'التعرّف'];

    let badFormulations = 0;
    lines.forEach((line, i) => {
        const startsGood = goodStarters.some(s => line.trim().startsWith(s));
        const startsBad = badStarters.some(s => line.trim().startsWith(s));

        if (startsBad) {
            badFormulations++;
        }
        if (!startsGood && !startsBad && line.trim().length > 5) {
            // قد يكون الهدف مسبوقاً برقم
            const cleanLine = line.replace(/^[\d\.\-\)]+\s*/, '').trim();
            const cleanStartsGood = goodStarters.some(s => cleanLine.startsWith(s));
            if (!cleanStartsGood) badFormulations++;
        }
    });

    if (badFormulations > 0) {
        score.deductions.push(badFormulations * 5);
        notes.push(makeNote('medium', 'صياغة بعض الأهداف غير قابلة للقياس', `${badFormulations} هدف(أهداف) تستخدم صياغات عامة غير قابلة للقياس المباشر.`, 'أعد صياغة الأهداف بأفعال قابلة للقياس مثل: تحديد، قياس، تقييم، تحليل، التحقق من...');
    }

    // فحص التكرار
    const objectiveTexts = lines.map(l => l.replace(/^[\d\.\-\)]+\s*/, '').trim());
    for (let i = 0; i < objectiveTexts.length; i++) {
        for (let j = i + 1; j < objectiveTexts.length; j++) {
            if (similarity(objectiveTexts[i], objectiveTexts[j]) > 0.7) {
                score.deductions.push(8);
                notes.push(makeNote('high', 'تكرار بين الأهداف', `الهدف "${objectiveTexts[i].substring(0, 40)}..." متكرر بشكل كبير مع هدف آخر.`, 'دمج الأهداف المتكررة أو تمييزها بوضوح.'));
                break;
            }
        }
    }
}

// --- تحليل الفرضيات ---
function analyzeHypotheses(text, score, notes) {
    const lines = text.split('\n').filter(l => l.trim().length > 5);

    if (lines.length < 1) {
        score.deductions.push(15);
        notes.push(makeNote('medium', 'لا توجد فرضيات واضحة', 'معظم الرسائل الكمية تحتاج فرضيات لتوجيه التحليل.', 'صغ فرضيات تتعلق بالعلاقات بين متغيرات الدراسة.');
    }

    // فحص صياغة الفرضيات
    const hypothesisPatterns = [
        /توجد.*علاقة/,
        /لا توجد.*علاقة/,
        /يوجد.*تأثير/,
        /لا يوجد.*تأثير/,
        /يوجد.*فروق/,
        /لا توجد.*فروق/,
        /توجد.*دلالة/,
        /لا توجد.*دلالة/
    ];

    let wellFormed = 0;
    lines.forEach(line => {
        const clean = line.replace(/^[\d\.\-\)]+\s*/, '').trim();
        if (hypothesisPatterns.some(p => p.test(clean))) {
            wellFormed++;
        }
    });

    if (lines.length > 0 && wellFormed === 0) {
        score.deductions.push(15);
        notes.push(makeNote('high', 'الفرضيات غير مصاغة بالصيغة الإحصائية المطلوبة', 'الفرضيات يجب أن تصاغ بشكل يسمح باختبارها إحصائياً (مثلاً: توجد علاقة ذات دلالة إحصائية...).', 'أعد صياغة الفرضيات باستخدام الصيغة الإحصائية: "توجد/لا توجد علاقة دالة إحصائياً عند مستوى دلالة (0.05) بين..."'));
    } else if (wellFormed < lines.length && lines.length > 0) {
        score.deductions.push(8);
        notes.push(makeNote('medium', 'بعض الفرضيات غير مصاغة بالشكل الإحصائي المناسب', `${lines.length - wellFormed} فرضية تحتاج إعادة صياغة.`, 'وحّد صياغة جميع الفرضيات وفق النمط الإحصائي المعتمد.'));
    }

    // فحص وجود فرضيات صفرية وبديلة
    const hasNull = /لا توجد|لا يوجد|لا توجد.*فروق|لا توجد.*علاقة|لا يوجد.*تأثير/.test(text);
    const hasAlternative = /توجد.*علاقة|يوجد.*تأثير|توجد.*فروق/.test(text) && !/لا توجد|لا يوجد/.test(text.match(/توجد.*علاقة|يوجد.*تأثير|توجد.*فروق/)?.[0] || '');

    if (lines.length >= 2 && !hasNull) {
        score.deductions.push(5);
        notes.push(makeNote('low', 'لا توجد فرضيات صفرية واضحة', 'يفضل ذكر الفرضية الصفرية (لا توجد علاقة/فروق) جنباً إلى جنب مع الفرضية البديلة.', 'أضف الفرضيات الصفرية المقابلة لكل فرضية بديلة.'));
    }
}

// --- تحليل الإطار النظري ---
function analyzeTheory(text, score, notes) {
    const words = countWords(text);
    const paragraphs = text.split(/\n\s*\n/).filter(p => p.trim().length > 20);

    // الحجم
    if (words < 300 && document.getElementById('degreeLevel').value === 'phd') {
        score.deductions.push(20);
        notes.push(makeNote('high', 'الإطار النظري قصير جداً لرسالة دكتوراه', `يحتوي على ${words} كلمة فقط. الإطار النظري للدكتوراه يحتاج عادةً 3000+ كلمة.`, 'وسّع الإطار النظري بتغطية النظريات والدراسات السابقة بشكل أعمق.');
    } else if (words < 200) {
        score.deductions.push(15);
        notes.push(makeNote('medium', 'الإطار النظري قصير', `يحتوي على ${words} كلمة.`, 'وسّع الإطار النظري ليغطي المتغيرات الأساسية والدراسات السابقة.');
    }

    // فحص وجود مراجع/دراسات سابقة
    const hasCitations = /\(\d{4}\)|\[\d+\]|\d{4}[مه]/.test(text) || /دراسة|باحث|أشار|أكد|توصل/.test(text);
    if (!hasCitations) {
        score.deductions.push(15);
        notes.push(makeNote('high', 'الإطار النظري يفتقر للتوثيق والدراسات السابقة', 'الإطار النظري يجب أن يستند إلى دراسات سابقة ونظريات علمية موثقة.', 'أضف إشارات للدراسات السابقة والنظريات ذات الصلة مع التوثيق المناسب.');
    }

    // فحص وجود مصطلحات تعريفية
    const hasDefinitions = /يُعرّف|تعريف|مفهوم|يُقصد|المقصود/.test(text);
    if (!hasDefinitions) {
        score.deductions.push(8);
        notes.push(makeNote('medium', 'لا توجد تعريفات إجرائية للمتغيرات', 'الإطار النظري يجب أن يتضمن تعريفات إجرائية للمتغيرات الرئيسية.', 'أضف تعريفات إجرائية واضحة لكل متغير من متغيرات الدراسة.'));
    }

    // فحص الفقرات
    if (paragraphs.length < 3 && words > 200) {
        score.deductions.push(5);
        notes.push(makeNote('low', 'الإطار النظري يحتاج تهيئة فقرات أفضل', `النص مقسم إلى ${paragraphs.length} فقرات فقط.`, 'قسّم الإطار النظري إلى فقرات مترابطة كل منها يغطي محوراً محدداً.');
    }

    // كثافة المصطلحات الأكاديمية
    const academicTerms = ['متغير', 'مستقل', 'تابع', 'وسيط', 'ضابط', 'نظرية', 'منهج', 'عينة', 'صدق', 'ثبات', 'إحصاء', 'دلالة', 'معامل', 'ارتباط', 'انحدار'];
    let academicCount = 0;
    academicTerms.forEach(t => { if (text.includes(t)) academicCount++; });
    if (academicCount < 2 && words > 200) {
        score.deductions.push(5);
        notes.push(makeNote('low', 'قلة المصطلحات الأكاديمية المتخصصة', 'الإطار النظري يجب أن يستخدم مصطلحات تخصصية دقيقة.', 'استخدم مصطلحات أكاديمية أكثر دقة وتخصصاً.');
    }
}

// --- تحليل المنهجية ---
function analyzeMethodology(text, score, notes) {
    // عناصر المنهجية المطلوبة
    const required = [
        { terms: ['منهج', 'المنهج', 'المنهجية', 'أسلوب'], name: 'نوع المنهج' },
        { terms: ['عينة', 'المجتمع', 'أفراد العينة', 'حجم العينة'], name: 'المجتمع والعينة' },
        { terms: ['أداة', 'استبانة', 'اختبار', 'مقياس', 'استبيان', 'مقابلة', 'ملف', 'بطاقة'], name: 'أداة البحث' },
        { terms: ['صدق', 'ثبات', 'معامل', 'كرونباخ', 'ألفا', 'Cronbach', 'صدق'], name: 'الصدق والثبات' },
        { terms: ['إحصائي', 'اختبار', 'SPSS', 'تحليل', 'معامل ارتباط', 'انحدار', 'T-test', 'ANOVA', 'تحليل التباين'], name: 'المعالجة الإحصائية' }
    ];

    let missing = [];
    required.forEach(r => {
        if (!r.terms.some(t => text.includes(t))) missing.push(r.name);
    });

    if (missing.length >= 3) {
        score.deductions.push(25);
        notes.push(makeNote('high', 'المنهجية ناقصة بشكل جوهري', `العناصر المفقودة: ${missing.join('، ')}`, 'أعد كتابة المنهجية لتشمل جميع العناصر الأساسية بالتفصيل.');
    } else if (missing.length > 0) {
        score.deductions.push(missing.length * 7);
        notes.push(makeNote('medium', 'المنهجية تفتقر لبعض العناصر الأساسية', `العناصر المفقودة: ${missing.join('، ')}`, 'أضف العناصر المفقودة مع التفصيل المناسب.');
    }

    // فحص ذكر حجم العينة بأرقام
    if (!/\d+/.test(text)) {
        score.deductions.push(10);
        notes.push(makeNote('medium', 'لا توجد أرقام محددة في المنهجية', 'المنهجية يجب أن تحتوي على أرقام دقيقة: حجم العينة، عدد فقرات الأداة، معاملات الصدق والثبات.', 'أضف الأرقام والإحصائيات اللازمة.');
    }

    // فحص نوع المنهج المحدد
    const methodTypes = ['وصفي', 'تجريبي', 'شبه تجريبي', 'تحليلي', 'تاريخي', 'ميداني', 'مسحي', 'تطبيقي', 'إرشادي', 'سلوكي'];
    const hasMethodType = methodTypes.some(m => text.includes(m));
    if (!hasMethodType) {
        score.deductions.push(10);
        notes.push(makeNote('high', 'لم يتم تحديد نوع المنهج بوضوح', 'يجب تحديد نوع المنهج المستخدم بشكل صريح (وصفي، تجريبي، تحليلي...).', 'حدد نوع المنهج المستخدم مع تبرير اختياره.');
    }

    // فحص معامل كرونباخ ألفا
    if (text.includes('ألفا') || text.includes('كرونباخ') || text.toLowerCase().includes('cronbach')) {
        const alphaMatch = text.match(/(?:ألفا|كرونباخ|Cronbach)[^\d]*(\d+\.?\d*)/);
        if (alphaMatch) {
            const alpha = parseFloat(alphaMatch[1]);
            if (alpha < 0.6) {
                score.deductions.push(12);
                notes.push(makeNote('high', `معامل كرونباخ ألفا منخفض (${alpha})`, 'المعقبول عادةً هو 0.70 فأعلى. القيمة المنخفضة تشير إلى مشاكل في ثبات الأداة.', 'راجع فقرات الأداة واحذف الفقرات الضعيفة أو أعد صياغتها.');
            } else if (alpha < 0.70) {
                score.deductions.push(5);
                notes.push(makeNote('low', `معامل كرونباخ ألفا مقبول بشكل محدود (${alpha})`, 'القيمة مقبولة لكنها قريبة من الحد الأدنى.', 'حاول تحسين الثبات بمراجعة الفقرات.'));
            }
        }
    }
}

// --- تحليل تحليل البيانات ---
function analyzeAnalysis(text, score, notes) {
    // فحص وجود اختبارات إحصائية
    const statTests = [
        { pattern: /T.test|t-test|اختبار ت/, name: 'اختبار t' },
        { pattern: /ANOVA|تحليل التباين|analysis of variance/i, name: 'تحليل التباين' },
        { pattern: /ارتباط|correlation|بيرسون|Pearson/i, name: 'معامل الارتباط' },
        { pattern: /انحدار|regression/i, name: 'الانحدار' },
        { pattern: /كاي مربع|Chi.square|chi-square/i, name: 'كاي مربع' },
        { pattern: /متوسط|وسط حسابي|mean/i, name: 'المتوسطات الحسابية' },
        { pattern: /انحراف معياري|standard deviation/i, name: 'الانحراف المعياري' }
    ];

    let foundTests = [];
    statTests.forEach(t => {
        if (t.pattern.test(text)) foundTests.push(t.name);
    });

    if (foundTests.length === 0 && countWords(text) > 30) {
        score.deductions.push(20);
        notes.push(makeNote('high', 'لا توجد اختبارات إحصائية واضحة', 'قسم تحليل البيانات يجب أن يعرض نتائج الاختبارات الإحصائية المستخدمة.', 'أضف جداول النتائج الإحصائية مع تفسيرها.');
    }

    // فحص وجود جداول أو أرقام
    const hasNumbers = (text.match(/\d+\.\d{2,}/g) || []).length;
    if (hasNumbers < 3 && countWords(text) > 50) {
        score.deductions.push(10);
        notes.push(makeNote('medium', 'قلة الأرقام الإحصائية في التحليل', 'التحليل الإحصائي يجب أن يحتوي على قيم إحصائية دقيقة (معاملات، مستويات دلالة، درجات حرية).', 'أضف القيم الإحصائية الكاملة لكل اختبار.');
    }

    // فحص مستوى الدلالة
    const hasSigLevel = /0\.05|0\.01|دلالة إحصائية|مستوى الدلالة|Sig/.test(text);
    if (!hasSigLevel && countWords(text) > 50) {
        score.deductions.push(8);
        notes.push(makeNote('medium', 'لم يتم ذكر مستوى الدلالة الإحصائية', 'يجب تحديد مستوى الدلالة المستخدم (عادةً 0.05 أو 0.01).', 'أضف بيان مستوى الدلالة المعتمد في البحث.');
    }

    // فحص التفسير
    const hasInterpretation = /أي أن|مما يعني|يشير هذا|يدل هذا|يعزى هذا|تفسر|بمعنى/.test(text);
    if (!hasInterpretation && countWords(text) > 100) {
        score.deductions.push(10);
        notes.push(makeNote('medium', 'التحليل يفتقر لتفسير النتائج الإحصائية', 'لا يكفي عرض الأرقام، بل يجب تفسيرها بلغة واضحة.', 'أضف تفسيراً لكل نتيجة إحصائية بما يفهمه القارئ غير المتخصص.');
    }
}

// --- تحليل النتائج ---
function analyzeResults(text, score, notes) {
    const words = countWords(text);

    if (words < 50) {
        score.deductions.push(20);
        notes.push(makeNote('high', 'قسم النتائج قصير جداً', 'النتائج تحتاج إلى تفصيل أكثر وربط واضح بالفرضيات.', 'وسّع قسم النتائج مع عرض كل نتيجة مرتبطة بفرضية محددة.');
    }

    // فحص الربط بالفرضيات
    const hasHypLink = /فرضية|الفرضية|الفرضية الأولى|الفرضية الثانية|الفرض/.test(text);
    if (!hasHypLink && words > 50) {
        score.deductions.push(12);
        notes.push(makeNote('high', 'النتائج غير مرتبطة بالفرضيات بوضوح', 'كل نتيجة يجب أن تُربط بالفرضية التي تتحقق منها.', 'أعد تنظيم النتائج بحيث كل فرضية لها نتيجة مقابلة.');
    }

    // فحص وجود كلمات النتائج
    const resultWords = /توصلت|أظهرت|أثبتت|تبين|أكدت|دلت|أسفرت|تبين أن|تشير النتائج/.test(text);
    if (!resultWords && words > 30) {
        score.deductions.push(8);
        notes.push(makeNote('medium', 'صياغة النتائج ضعيفة', 'النتائج يجب أن تُصاغ بصيغة جازمة واضحة.', 'استخدم صياغات مثل: "توصلت الدراسة إلى..." أو "أظهرت النتائج أن..."'));
    }

    // فحص التدرج المنطقي
    const hasTransition = /أما|كما|وبالتالي|إضافة إلى|فضلاً عن|علاوة على|فيما يتعلق/.test(text);
    if (!hasTransition && words > 100) {
        score.deductions.push(5);
        notes.push(makeNote('low', 'ضعف في الترابط بين النتائج', 'استخدم أدوات الربط لتحقيق التسلسل المنطقي بين النتائج.', 'أضف أدوات ربط مناسبة بين النتائج المختلفة.'));
    }
}

// --- تحليل التوصيات ---
function analyzeRecommendations(text, score, notes) {
    const lines = text.split('\n').filter(l => l.trim().length > 5);

    if (lines.length < 2) {
        score.deductions.push(15);
        notes.push(makeNote('medium', 'عدد التوصيات قليل', 'يجب تقديم توصيات كافية تستند إلى النتائج (3-8 توصيات).', 'أضف توصيات إضافية تنبثق من نتائج الدراسة.');
    }

    // فحص صياغة التوصيات
    const goodRecPatterns = [/يوصي/, /يُقترح/, /يجب/, /ينبغي/, /من الضروري/, /يفضل/];
    let badRecs = 0;
    lines.forEach(line => {
        const clean = line.replace(/^[\d\.\-\)]+\s*/, '').trim();
        if (!goodRecPatterns.some(p => p.test(clean)) && clean.length > 10) {
            badRecs++;
        }
    });

    if (badRecs > 0 && lines.length > 0) {
        score.deductions.push(badRecs * 4);
        notes.push(makeNote('medium', 'بعض التوصيات غير مصاغة بصيغة التوصية', `${badRecs} توصية لا تبدأ بصيغة توصية واضحة.`, 'ابدأ كل توصية بـ: "يوصي الباحث بـ..." أو "يُقترح..."'));
    }

    // فحص الارتباط بالنتائج
    const hasResultLink = /بناءً على|استناداً إلى|وفقاً لنتائج|في ضوء نتائج|بناء على نتائج/.test(text);
    if (!hasResultLink && lines.length >= 2) {
        score.deductions.push(8);
        notes.push(makeNote('medium', 'التوصيات غير مرتبطة بوضوح بنتائج الدراسة', 'كل توصية يجب أن تستند إلى نتيجة محددة من الدراسة.', 'اربط كل توصية بالنتيجة التي تستند إليها.');
    }
}

// --- تحليل المراجع ---
function analyzeReferences(text, score, notes) {
    const lines = text.split('\n').filter(l => l.trim().length > 10);

    if (lines.length < 5) {
        score.deductions.push(20);
        notes.push(makeNote('high', 'عدد المراجع قليل جداً', `يوجد ${lines.length} مرجع فقط. الحد الأدنى عادةً 20-30 مرجعاً للماجستير و40-60 للدكتوراه.`, 'أضف المزيد من المراجع الحديثة والمتنوعة.');
    }

    // فحص حداثة المراجع
    let oldRefs = 0;
    let recentRefs = 0;
    let noYearRefs = 0;
    lines.forEach(line => {
        const years = line.match(/(?:19|20)\d{2}/g);
        if (!years) { noYearRefs++; return; }
        years.forEach(y => {
            if (parseInt(y) < 2010) oldRefs++;
            else recentRefs++;
        });
    });

    if (oldRefs > recentRefs && lines.length > 5) {
        score.deductions.push(12);
        notes.push(makeNote('medium', 'معظم المراجع قديمة', `المراجع القديمة (قبل 2010) أكثر من الحديثة. يُفضل أن تكون غالبية المراجع خلال آخر 10 سنوات.`, 'حدّث قائمة المراجع بإضافة مصادر حديثة.');
    }

    if (noYearRefs > 0) {
        score.deductions.push(noYearRefs * 3);
        notes.push(makeNote('medium', `${noYearRefs} مرجع بدون سنة نشر`, 'كل مرجع يجب أن يتضمن سنة النشر.', 'أكمل بيانات المراجع الناقصة.');
    }

    // فحص تنوع المصادر
    const hasArabic = lines.some(l => /[\u0600-\u06FF]/.test(l));
    const hasEnglish = lines.some(l => /[a-zA-Z]{3,}/.test(l));
    if (!hasEnglish && lines.length > 10) {
        score.deductions.push(8);
        notes.push(makeNote('low', 'لا توجد مراجع أجنبية', 'التنوع في المصادر (عربية وأجنبية) يعزز مصداقية البحث.', 'أضف مراجع أجنبية حديثة ذات صلة.');
    }

    // فحص التكرار
    const refTexts = lines.map(l => l.replace(/\s+/g, ' ').trim());
    for (let i = 0; i < refTexts.length; i++) {
        for (let j = i + 1; j < refTexts.length; j++) {
            if (similarity(refTexts[i], refTexts[j]) > 0.8) {
                score.deductions.push(5);
                notes.push(makeNote('medium', 'مرجع مكرر في قائمة المراجع', `المرجع: "${refTexts[i].substring(0, 50)}..." مكرر.`, 'احذف المراجع المكررة.');
                break;
            }
        }
    }

    // فحص أسلوب التوثيق
    const hasConsistentStyle = (text.match(/\(\d{4}\)/g) || []).length > 2 ||
                                (text.match(/\[\d+\]/g) || []).length > 2;
    if (!hasConsistentStyle && lines.length > 5) {
        score.deductions.push(10);
        notes.push(makeNote('medium', 'أسلوب التوثيق غير واضح أو غير موحد', 'يجب اتباع أسلوب توثيق موحد (APA، MLA، Chicago...).', 'وحّد أسلوب التوثيق في جميع المراجع وفق الدليل المعتمد.');
    }
}

// ============================================
// فحوصات نصية عامة
// ============================================
function applyGeneralTextChecks(text, score, notes, sectionKey) {
    // كلمات عامية
    checkColloquialWords(text, score, notes);

    // جمل طويلة جداً
    const sentences = text.split(/[.！؟\n]+/).filter(s => s.trim().length > 0);
    let longSentences = 0;
    sentences.forEach(s => {
        const words = s.trim().split(/\s+/).length;
        if (words > 40) longSentences++;
    });
    if (longSentences > 2) {
        score.deductions.push(longSentences * 2);
        notes.push(makeNote('low', `${longSentences} جملة طويلة جداً (أكثر من 40 كلمة)`, 'الجمل الطويلة جداً تصعب الفهم وتضعف وضوح النص.', 'قسّم الجمل الطويلة إلى جمل أقصر وأوضح.'));
    }

    // تكرار الكلمات المفرط
    const wordFreq = {};
    const cleanWords = text.replace(/[^\u0600-\u06FFa-zA-Z\s]/g, '').split(/\s+/).filter(w => w.length > 3);
    cleanWords.forEach(w => {
        w = w.trim();
        if (w.length > 3) wordFreq[w] = (wordFreq[w] || 0) + 1;
    });

    const stopWords = new Set(['التي', 'الذي', 'عليه', 'فيها', 'منها', 'عنها', 'إليه', 'معها', 'بها', 'لها', 'ذلك', 'هذه', 'هذا', 'تلك', 'كانت', 'يكون', 'يمكن', 'كما', 'بشكل', 'وذلك', 'وقد', 'حيث', 'خلال']);
    let overused = [];
    for (let w in wordFreq) {
        if (!stopWords.has(w) && wordFreq[w] > Math.max(5, cleanWords.length * 0.05)) {
            overused.push({ word: w, count: wordFreq[w] });
        }
    }
    if (overused.length > 0) {
        const top3 = overused.sort((a,b) => b.count - a.count).slice(0, 3);
        score.deductions.push(top3.length * 3);
        notes.push(makeNote('low', 'تكرار مفرط لبعض الكلمات', `الكلمات الأكثر تكراراً: ${top3.map(t => `"${t.word}" (${t.count} مرة)`).join('، ')}`, 'استخدم مرادفات لتقليل التكرار وتحسين تنوع المفردات.'));
    }
}

function checkColloquialWords(text, score, notes) {
    const colloquial = [
        'يعني', 'طيب', 'حلو', 'زين', 'كذا', 'عشان', 'ليش', 'شو', 'ايش', 'يقول',
        'بس', 'هذي', 'هذا الكلام', 'بصراحة', 'بجد', 'تمام', 'طيب', 'اللي',
        'عندي', 'الحين', 'يلا', 'خلّ', 'لازم', 'ممكن', 'أبغى', 'ابغى'
    ];

    let found = [];
    colloquial.forEach(w => {
        if (text.includes(w)) found.push(w);
    });
    if (found.length > 0) {
        score.deductions.push(found.length * 3);
        notes.push(makeNote('medium', 'وجود كلمات عامية في النص الأكاديمي', `الكلمات: ${found.map(w => `"${w}"`).join('، ')}`, 'استبدل الكلمات العامية بمصطلحات أكاديمية رصينة.'));
    }
}

// ============================================
// ملاحظات عامة
// ============================================
function generateGeneralNotes(filledSections, results) {
    const notes = [];
    const degree = document.getElementById('degreeLevel').value;
    const roleLabels = {
        supervisor: 'مشرف الرسالة',
        internal: 'المناقش الداخلي',
        external: 'المناقش الخارجي',
        researcher: 'الباحث'
    };

    // فحص الترابط بين الأقسام
    const hasObjectives = filledSections.includes('objectives');
    const hasHypotheses = filledSections.includes('hypotheses');
    const hasResults = filledSections.includes('results');

    if (hasObjectives && hasResults) {
        notes.push(makeNote('info', 'ملاحظة حول الترابط بين الأهداف والنتائج', 'تأكد من أن كل هدف في الرسالة له نتيجة مقابلة تؤكد تحقيقه أو عدم تحقيقه. هذا من أهم معايير جودة الرسالة.', 'راجع جدول المطابقة بين الأهداف والنتائج والتوصيات.'));
    }

    if (hasHypotheses && hasResults) {
        notes.push(makeNote('info', 'ملاحظة حول الترابط بين الفرضيات والنتائج', 'تأكد من أن كل فرضية تم اختبارها وعرضت نتيجتها بشكل صريح (مقبولة/مرفوضة).', 'أنشئ جدولاً يلخص نتائج اختبار كل فرضية.'));
    }

    // فحص حسب الدرجة
    if (degree === 'phd') {
        const hasTheory = filledSections.includes('theory');
        if (hasTheory && countWords(sectionData.theory || '') < 1000) {
            notes.push(makeNote('medium', 'الإطار النظري لرسالة دكتوراه يحتاج توسيعاً كبيراً', 'رسالة الدكتوراه تتطلب إطاراً نظرياً عميقاً ومفصلاً يتضمن نقداً للدراسات السابقة ومساهمة نظرية واضحة.', 'وسّع الإطار النظري مع التركيز على المساهمة النظرية للدراسة.'));
        }
    }

    // ملاحظة حسب الدور
    if (currentRole === 'supervisor') {
        notes.push(makeNote('info', 'توصية للمشرف', 'كمشرف، ركز على التوجيه نحو تصحيح الفجوات المنهجية والنظرية قبل تقديم الرسالة للمناقشة. تأكد من连贯ة السرد الأكاديمي بين جميع الأقسام.', 'اعقد اجتماعاً مع الباحث لمناقشة الملاحظات ذات الأولوية العالية أولاً.'));
    } else if (currentRole === 'external') {
        notes.push(makeNote('info', 'توصية للمناقش الخارجي', 'التركيز على: أصالة العنوان، عمق الإطار النظري، ومنهجية البحث. قيّم مدى إضافة الرسالة للمعرفة العلمية في التخصص.', 'جهّز أسئلة المناقشة بناءً على الفجوات المكتشفة.'));
    }

    // فحص الكلمات المكررة بشكل عام
    if (results.stats.totalWords > 200) {
        const sorted = Object.entries(results.wordFrequency)
            .filter(([w]) => w.length > 3 && !new Set(['التي','الذي','عليه','فيها','منها','عنها','إليه','معها','بها','لها','ذلك','هذه','هذا','تلك','كانت','يكون','يمكن','كما','بشكل','وذلك','وقد','حيث','خلال','على','من','إلى','في','عن','مع','ذلك','هذا','بأن','التى']).has(w))
            .sort((a,b) => b[1] - a[1])
            .slice(0, 8);

        if (sorted.length > 0) {
            const cloud = sorted.map(([w, c]) => {
                const ratio = c / (results.stats.totalWords * 0.01);
                let cls = 'normal';
                if (ratio > 3) cls = 'overused';
                else if (ratio > 1.5) cls = 'frequent';
                return `<span class="word-tag ${cls}">${w} (${c})</span>`;
            }).join('');
            notes.push({
                severity: 'info',
                title: 'تحليل تكرار الكلمات الأكثر شيوعاً',
                body: `<div class="words-cloud">${cloud}</div>`,
                suggestion: null
            });
        }
    }

    return notes;
}

// ============================================
// أدوات مساعدة

function makeNote(severity, title, body, suggestion) {
    return { severity, title, body, suggestion };
}

function countWords(text) {
    return text.split(/\s+/).filter(w => w.length > 0).length;
}

function similarity(a, b) {
    const wordsA = new Set(a.split(/\s+/));
    const wordsB = new Set(b.split(/\s+/));
    const intersection = [...wordsA].filter(w => wordsB.has(w)).length;
    const union = new Set([...wordsA, ...wordsB]).size;
    return union > 0 ? intersection / union : 0;
}

function extractWordFrequency(text, freqMap) {
    const clean = text.replace(/[^\u0600-\u06FFa-zA-Z\s]/g, '');
    clean.split(/\s+/).filter(w => w.length > 2).forEach(w => {
        w = w.trim();
        if (w.length > 2) freqMap[w] = (freqMap[w] || 0) + 1;
    });
}

function getScoreColor(score) {
    if (score >= 80) return 'var(--green)';
    if (score >= 60) return 'var(--orange)';
    return 'var(--red)';
}

function getScoreLabel(score) {
    if (score >= 90) return 'ممتاز';
    if (score >= 80) return 'جيد جداً';
    if (score >= 70) return 'جيد';
    if (score >= 60) return 'مقبول';
    if (score >= 50) return 'ضعيف';
    return 'ضعيف جداً';
}

function getSeverityLabel(s) {
    const map = { high: 'حرج', medium: 'متوسط', low: 'ثانوي', info: 'معلوماتي' };
    return map[s] || s;
}

function animateNumber(elId, from, to, duration) {
    const el = document.getElementById(elId);
    const start = performance.now();
    function update(now) {
        const progress = Math.min((now - start) / duration, 1);
        const eased = 1 - Math.pow(1 - progress, 3);
        el.textContent = Math.round(from + (to - from) * eased);
        if (progress < 1) requestAnimationFrame(update);
    }
    requestAnimationFrame(update);
}

function renderSectionReports(r) {
    const container = document.getElementById('sectionReports');
    let html = '';

    for (let key in r.sectionScores) {

        const score = r.sectionScores[key];
        const notes = r.sectionNotes[key];
        const color = getScoreColor(score);

        html += `

                <div class="section-report-header">
                    <div class="section-icon" style="background:${sec.color}22; color:${sec.color}">
                        <i class="${sec.icon}"></i>
                    </div>
                    <h3>${sec.label}</h3>
                    <div class="section-score" style="color:${color}">${score}%</div>
                </div>
                <div class="criteria-list">
        `;

        // معايير النجاح/الفشل
        const criteria = buildCriteria(key, score, notes);
        criteria.forEach(c => {
            html += `
                <div class="criteria-item">
                    <div class="crit-icon ${c.status}">
                        <i class="fas ${c.status === 'pass' ? 'fa-check' : c.status === 'fail' ? 'fa-times' : 'fa-exclamation'}"></i>
                    </div>
                    <div class="crit-text">${c.text}</div>
                </div>
            `;
        });

        html += `</div></div>`;
    }

    container.innerHTML = html;
}

function renderGeneralNotes(r) {
    const container = document.getElementById('generalNotes');
    if (!r.generalNotes || r.generalNotes.length === 0) {
        container.classList.remove('show');
        return;
    }

    // تجميع الملاحظات من جميع الأقسام أيضاً
    let allNotes = [...r.generalNotes];
    for (let key in r.sectionNotes) {
        r.sectionNotes[key].forEach(n => {
            allNotes.push({
                ...n,
                section: SECTIONS[key].label
            });
        });
    }

    // ترتيب حسب الأهمية
    const order = { high: 0, medium: 1, low: 2, info: 3 };
    allNotes.sort((a, b) => (order[a.severity] || 3) - (order[b.severity] || 3));

    let html = `
        <div class="notes-section-header">
            <h3><i class="fas fa-clipboard-list" style="color:var(--accent)"></i> جميع الملاحظات التفصيلية</h3>
            <span class="badge" style="background:var(--accent-dim); color:var(--accent)">${allNotes.length} ملاحظة</span>
        </div>
    `;

    allNotes.forEach((n, i) => {
        html += `
            <div class="note-card severity-${n.severity}" style="animation: fadeUp 0.4s ease ${i * 0.05}s both">
                <div class="note-card-head">
                    <span class="severity-dot ${n.severity}"></span>
                    <span class="note-title">${n.section ? `[${n.section}] ` : ''}${n.title}</span>
                    <span class="severity-badge ${n.severity}">${getSeverityLabel(n.severity)}</span>
                </div>
                <div class="note-card-body">${n.body}</div>
                ${n.suggestion ? `<div class="note-suggestion"><i class="fas fa-arrow-left"></i><span>${n.suggestion}</span></div>` : ''}
            </div>
        `;
    });

    container.innerHTML = html;
    container.classList.add('show');
}

function exportReport(type) {
    if (!analysisResults) return;
    const r = analysisResults;
    const degree = document.getElementById('degreeLevel').value === 'masters' ? 'ماجستير' : 'دكتوراه';
    const field = document.getElementById('fieldOfStudy').value || 'غير محدد';

    let text = `═══════════════════════════════════════\n`;
    text += `   تقرير مراجعة الرسالة الأكاديمية\n`;
    text += `═══════════════════════════════════════\n\n`;
    text += `الدرجة العلمية: ${degree}\n`;
    text += `التخصص: ${field}\n`;
    text += `التقييم العام: ${r.totalScore}% - ${getScoreLabel(r.totalScore)}\n`;
    text += `عدد الأقسام المحللة: ${r.stats.sectionsAnalyzed}\n`;
    text += `إجمالي الكلمات: ${r.stats.totalWords}\n`;
    text += `الملاحظات الحرجة: ${r.stats.highIssues}\n`;
    text += `الملاحظات المتوسطة: ${r.stats.mediumIssues}\n`;
    text += `الملاحظات الثانوية: ${r.stats.lowIssues}\n\n`;

    text += `───────────────────────────────────────\n`;
    text += `   درجات الأقسام\n`;
    text += `───────────────────────────────────────\n\n`;

    for (let key in r.sectionScores) {
        const sec = SECTIONS[key];
        text += `● ${sec.label}: ${r.sectionScores[key]}% (${getScoreLabel(r.sectionScores[key])})\n`;
    }

    text += `\n───────────────────────────────────────\n`;
    text += `   الملاحظات التفصيلية\n`;
    text += `───────────────────────────────────────\n\n`;

    for (let key in r.sectionNotes) {
        const sec = SECTIONS[key];
        r.sectionNotes[key].forEach(n => {
            text += `[${sec.label}] [${getSeverityLabel(n.severity)}] ${n.title}\n`;
            text += `  ${n.body.replace(/<[^>]+>/g, '')}\n`;
            if (n.suggestion) text += `  >> المقترح: ${n.suggestion}\n`;
            text += `\n`;
        });
    }

    if (r.generalNotes.length > 0) {
        text += `───────────────────────────────────────\n`;
        text += `   ملاحظات عامة\n`;
        text += `───────────────────────────────────────\n\n`;
        r.generalNotes.forEach(n => {
            text += `[${getSeverityLabel(n.severity)}] ${n.title}\n`;
            text += `  ${n.body.replace(/<[^>]+>/g, '')}\n\n`;
        });
    }

    text += `\n═══════════════════════════════════════\n`;
    text += `تم إنشاء هذا التقرير بواسطة منصة مراجعة الرسائل الأكاديمية\n`;
    text += `التاريخ: ${new Date().toLocaleDateString('ar-EG')}\n`;
    text += `═══════════════════════════════════════\n`;

    // تحميل الملف
    const blob = new Blob([text], { type: 'text/plain;charset=utf-8' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `تقرير_مراجعة_الرسالة_${new Date().toISOString().split('T')[0]}.txt`;
    a.click();
    URL.revokeObjectURL(url);

    showToast('تم تصدير التقرير بنجاح', 'success');
}

function showToast(message, type = 'success') {
    const container = document.getElementById('toastContainer');
    const toast = document.createElement('div');
    toast.className = `toast ${type}`;
    const icons = { success: 'fa-check-circle', error: 'fa-exclamation-circle', warning: 'fa-exclamation-triangle' };
    toast.innerHTML = `<i class="fas ${icons[type] || icons.success}"></i> ${message}`;
    container.appendChild(toast);
    setTimeout(() => { toast.remove(); }, 3200);
}
