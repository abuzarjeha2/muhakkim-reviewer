def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in {'pdf', 'docx'}

def extract_text(file_path):
    ext = file_path.rsplit('.', 1)[1].lower()
    text = ""
    if ext == 'pdf':
        with open(file_path, 'rb') as f:
            for page in PyPDF2.PdfReader(f).pages:
                t = page.extract_text()
                if t: text += t + "\n"
    elif ext == 'docx':
        for para in docx.Document(file_path).paragraphs:
            text += para.text + "\n"
    return text

def chunk_text(text, chunk_size=6000, overlap=400):
    chunks, start = [], 0
    while start < len(text):
        end = min(start + chunk_size, len(text))
        chunk = text[start:end]
        # قص ذكي عند نهاية جملة أو فقرة
        cut = max(chunk.rfind('.'), chunk.rfind('\n'), chunk.rfind('؟'), chunk.rfind('!'))
        if cut > chunk_size * 0.4:
            chunk = chunk[:cut+1]
            end = start + cut + 1
        chunks.append(chunk)
        start = end - overlap
    return chunks or [text]

def clean_json(text):
    m = re.search(r'```(?:json)?\s*([\s\S]*?)\s*```', text)
    raw = m.group(1).strip() if m else text.strip()
    raw = raw.lstrip(',').rstrip(',')
    try: return json.loads(raw)
    except: return json.loads(raw.replace("'", '"'))

def call_ai(prompt, tokens=4000):
    resp = client.chat.completions.create(
        model=MODEL_NAME,
        messages=[{"role": "system", "content": prompt}],
        temperature=0.2, max_tokens=tokens
    )
    return clean_json(resp.choices[0].message.content)

@app.route('/')
def index():
    return render_template_string(HTML_TEMPLATE)

@app.route('/review', methods=['POST'])
def review():
    if 'file' not in request.files: return jsonify({"error": "لم يتم رفع ملف"}), 400
    f = request.files['file']
    if not allowed_file(f.filename): return jsonify({"error": "نوع غير مدعوم (PDF/DOCX)"}), 400

    with tempfile.NamedTemporaryFile(delete=False, suffix=f".{f.filename.rsplit('.',1)[1]}") as tmp:
        f.save(tmp.name)
        text = extract_text(tmp.name)
    os.unlink(tmp.name)
    if len(text.strip()) < 200: return jsonify({"error": "نص غير كافٍ أو ملف فارغ"}), 400

    chunks = chunk_text(text)
    sys_prompt = """أنت خبير أكاديمي. راجع الجزء المرفق وأرجع JSON فقط بالعربية:
    {"title_review":"","toc_review":"","abstract_review":"","theoretical_framework_review":"","methodology_review":"","data_analysis_review":"","results_review":"","recommendations_review":"","hypotheses_objectives_review":"","gaps":"","score":0}
    إذا غاب قسم، اكتب 'غير موجود'."""

    partials = []
    for i, c in enumerate(chunks):
        partials.append(call_ai(f"{sys_prompt}\n\nجزء {i+1}/{len(chunks)}:\n{c}"))

    merge_prompt = f"""أنت منسق أكاديمي. لديك {len(partials)} مراجعات جزئية.
    دمجهما في تقرير موحد، احذف التكرار، وحسّن الصياغة. أرجع JSON مطابق للهيكل الأصلي.
    البيانات:\n{json.dumps(partials, ensure_ascii=False)}"""
    
    final = call_ai(merge_prompt, tokens=3000)

    report_id = uuid.uuid4().hex[:8]
    pdf_path = os.path.join(REPORTS_DIR, f"report_{report_id}.pdf")
    gen_pdf(final, pdf_path)

    return jsonify({"status": "success", "pdf_url": f"/download/{report_id}"})

def gen_pdf(data, out_path):

              "theoretical_framework_review":"الإطار النظري","methodology_review":"المنهجية",
              "data_analysis_review":"تحليل البيانات","results_review":"النتائج",
              "recommendations_review":"التوصيات","hypotheses_objectives_review":"الأهداف والفرضيات","gaps":"الفجوات المكتشفة"}
    body = "".join([f'<div class="sec"><h3>{v}</h3><p>{data.get(k,"")}</p></div>' 
                    for k,v in fields.items() if data.get(k) and data.get(k)!="غير موجود"])
    
    html = f"""<!DOCTYPE html><html lang="ar" dir="rtl"><head><meta charset="UTF-8">
    <style>

        body {{ font-family: 'Segoe UI', 'Noto Sans Arabic', Arial, sans-serif; direction: rtl; line-height: 1.6; color: #1e293b; }}
        h1 {{ text-align: center; color: #0f172a; border-bottom: 2px solid #2563eb; padding-bottom: 10px; }}
        .sec {{ margin-bottom: 14px; padding: 10px; background: #f8fafc; border-right: 3px solid #2563eb; border-radius: 4px; }}
        h3 {{ margin: 0 0 5px; color: #2563eb; font-size: 15px; }}
        p {{ margin: 0; font-size: 13px; text-align: justify; }}
        .ft {{ text-align: center; font-size: 10px; color: #64748b; margin-top: 30px; border-top: 1px solid #cbd5e1; padding-top: 10px; }}
        .sc {{ text-align: center; font-size: 18px; font-weight: bold; color: #059669; margin: 20px 0; }}
    </style></head><body>
    <h1>📊 تقرير المراجعة الأكاديمية الذكي</h1>
    <p style="text-align:center;color:#64748b">تاريخ: {datetime.now().strftime('%Y-%m-%d %H:%M')}</p>
