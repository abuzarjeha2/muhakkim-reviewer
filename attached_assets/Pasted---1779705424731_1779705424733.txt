لا يمكن تشغيل المشروع من ملف واحد، ولكني سأعطيك نصاً واحداً يحتوي على جميع الملفات الثلاثة مع فواصل واضحة، يمكنك نسخه كاملاً، ثم إنشاء الملفات بنفسك يدوياً حسب الشرح الموجود داخل النص.

انسخ النص التالي بالكامل، واحفظه في ملف مثلاً project_export.txt، ثم اتبع التعليمات داخله:

import os
import json
from flask import Flask, render_template, request, jsonify
import openai
from dotenv import load_dotenv

load_dotenv()
app = Flask(__name__)

# ضع مفتاح API الخاص بك هنا أو في ملف .env
openai.api_key = os.getenv("OPENAI_API_KEY")

def review_section(section_name, section_text, custom_prompt):
    if not section_text or section_text.strip() == "":
        return "لم يتم إدخال نص لهذا القسم."
    
    prompt = f"""أنت خبير أكاديمي متخصص في مراجعة رسائل الماجستير والدكتوراه. 
المطلوب: مراجعة قسم "{section_name}" التالي وتقديم ملاحظات دقيقة حول:
- الأخطاء المنهجية والمحتوى
- الفجوات أو النواقص
- نقاط الضعف والقوة
- اقتراحات للتحسين
يُرجى أن تكون الملاحظات واضحة ومحددة.

{custom_prompt}

نص القسم:
{section_text}

الملاحظات:
"""
    
    try:
        response = openai.ChatCompletion.create(
            model="gpt-3.5-turbo",
            messages=[
                {"role": "system", "content": "أنت أستاذ جامعي خبير في الإشراف على رسائل الدراسات العليا."},
                {"role": "user", "content": prompt}
            ],
            temperature=0.5,
            max_tokens=1000
        )
        return response.choices[0].message['content'].strip()
    except Exception as e:
        return f"حدث خطأ أثناء مراجعة القسم: {str(e)}"

@app.route('/')
def index():
    return render_template('index.html')

@app.route('/review', methods=['POST'])
def review():
    data = request.json
    
    sections = {
        "العنوان": {
            "text": data.get("title", ""),
            "prompt": "راجع دقة العنوان ومدى تعبيره عن محتوى الرسالة، واقترح تحسينات إن وجدت."
        },
        "الفهرس": {
            "text": data.get("index", ""),
            "prompt": "راجع تنظيم الفهرس وتسلسل الموضوعات، وهل يعكس هيكل الرسالة بشكل منطقي؟"
        },
        "المستخلص": {
            "text": data.get("abstract", ""),
            "prompt": "راجع شمولية المستخلص: المشكلة، الأهداف، المنهج، النتائج، والتوصيات. حدد الفجوات."
        },
        "الإطار النظري": {
            "text": data.get("theoretical", ""),
            "prompt": "راجع تغطية الأدبيات السابقة، حدد الثغرات النظرية، وتقييم تكامل النظريات مع موضوع الرسالة."
        },
        "المنهجية": {
            "text": data.get("methodology", ""),
            "prompt": "راجع تصميم البحث، أدوات جمع البيانات، العينة، الأساليب الإحصائية، ومدى ملاءمتها للإجابة عن الأسئلة."
        },
        "تحليل البيانات": {
            "text": data.get("analysis", ""),
            "prompt": "راجع دقة التحليل، وضوح النتائج الإحصائية أو النوعية، وتطابقها مع أسئلة البحث."
        },
        "النتائج": {
            "text": data.get("results", ""),
            "prompt": "راجع عرض النتائج وارتباطها بالفرضيات والأهداف، ومدى تفسيرها بشكل غير منحاز."
        },
        "التوصيات": {
            "text": data.get("recommendations", ""),
            "prompt": "راجع مدى واقعية التوصيات وارتباطها بنتائج الدراسة، وأهميتها للتطبيق أو الأبحاث المستقبلية."
        },
        "الفرضيات": {
            "text": data.get("hypotheses", ""),
            "prompt": "راجع صياغة الفرضيات، قابليتها للاختبار، واتساقها مع المنهجية."
        },
        "الأهداف": {
            "text": data.get("objectives", ""),
            "prompt": "راجع وضوح الأهداف وقابليتها للتحقق، ومدى تغطيتها لمشكلة البحث."
        }
    }
    
    results = {}
    for section_name, section_info in sections.items():
        review_text = review_section(section_name, section_info["text"], section_info["prompt"])
        results[section_name] = review_text
    
    return jsonify(results)

if __name__ == '__main__':
    app.run(debug=True)

<script>
    const sections = [
        { id: "title", label: "🏷️ العنوان" }, { id: "index", label: "📑 الفهرس / المحتويات" },
        { id: "abstract", label: "📄 المستخلص" }, { id: "theoretical", label: "📚 الإطار النظري" },
        { id: "methodology", label: "🔬 المنهجية" }, { id: "analysis", label: "📊 تحليل البيانات" },
        { id: "results", label: "📈 النتائج" }, { id: "recommendations", label: "💡 التوصيات" },
        { id: "hypotheses", label: "❓ الفرضيات" }, { id: "objectives", label: "🎯 الأهداف" }
    ];
    function buildForm() {
        const container = document.getElementById('sectionsContainer');
        container.innerHTML = '';
        sections.forEach(section => {
            const card = document.createElement('div');
            card.className = 'card';
            card.innerHTML = `<div class="card-header">${section.label}</div><div class="card-body"><textarea id="txt_${section.id}" placeholder="أدخل نص ${section.label} هنا..."></textarea><button class="review-btn" data-section="${section.id}">📌 مراجعة هذا القسم فقط</button><div id="result_${section.id}" class="review-result">بانتظار المراجعة...</div></div>`;
            container.appendChild(card);
        });
        document.querySelectorAll('.review-btn').forEach(btn => {
            btn.addEventListener('click', (e) => {
                reviewSingleSection(btn.getAttribute('data-section'));
            });
        });
    }
    async function reviewSingleSection(sectionId) {
        const text = document.getElementById(`txt_${sectionId}`).value;
        const resultDiv = document.getElementById(`result_${sectionId}`);
        resultDiv.innerHTML = "⏳ جاري المراجعة...";
        const payload = {};
        sections.forEach(sec => { payload[sec.id] = (sec.id === sectionId) ? text : ""; });
        try {
            const response = await fetch('/review', {
                method: 'POST', headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ title: payload.title, index: payload.index, abstract: payload.abstract, theoretical: payload.theoretical, methodology: payload.methodology, analysis: payload.analysis, results: payload.results, recommendations: payload.recommendations, hypotheses: payload.hypotheses, objectives: payload.objectives })
            });
            const data = await response.json();
            const map = { title:"العنوان", index:"الفهرس", abstract:"المستخلص", theoretical:"الإطار النظري", methodology:"المنهجية", analysis:"تحليل البيانات", results:"النتائج", recommendations:"التوصيات", hypotheses:"الفرضيات", objectives:"الأهداف" };
            if(data[map[sectionId]]) resultDiv.innerHTML = data[map[sectionId]].replace(/\n/g, '<br>');
            else resultDiv.innerHTML = "لا توجد ملاحظات.";
        } catch(err) { resultDiv.innerHTML = "خطأ: "+err.message; }
    }
    async function reviewAllSections() {
        document.getElementById('statusMsg').innerHTML = "🔄 جاري مراجعة جميع الأقسام...";
        const payload = {};
        for(const sec of sections) payload[sec.id] = document.getElementById(`txt_${sec.id}`).value;
        try {
            const response = await fetch('/review', { method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({ title:payload.title, index:payload.index, abstract:payload.abstract, theoretical:payload.theoretical, methodology:payload.methodology, analysis:payload.analysis, results:payload.results, recommendations:payload.recommendations, hypotheses:payload.hypotheses, objectives:payload.objectives }) });
            const data = await response.json();
            const map = { title:"العنوان", index:"الفهرس", abstract:"المستخلص", theoretical:"الإطار النظري", methodology:"المنهجية", analysis:"تحليل البيانات", results:"النتائج", recommendations:"التوصيات", hypotheses:"الفرضيات", objectives:"الأهداف" };
            for(const sec of sections) {
                const div = document.getElementById(`result_${sec.id}`);
                if(data[map[sec.id]]) div.innerHTML = data[map[sec.id]].replace(/\n/g, '<br>');
                else div.innerHTML = "لا توجد ملاحظات.";
            }
            document.getElementById('statusMsg').innerHTML = "✅ اكتملت المراجعة.";
        } catch(err) { document.getElementById('statusMsg').innerHTML = "❌ فشل: "+err.message; }
    }
    function resetAllFields() {
        for(const sec of sections) { document.getElementById(`txt_${sec.id}`).value = ''; document.getElementById(`result_${sec.id}`).innerHTML = 'بانتظار المراجعة...'; }
        document.getElementById('statusMsg').innerHTML = '';
    }
    window.onload = () => { buildForm(); document.getElementById('reviewAllBtn').addEventListener('click', reviewAllSections); document.getElementById('resetBtn').addEventListener('click', resetAllFields); };
