import { Toaster } from "@/components/ui/sonner";

import Home from "./pages/Home";
import About from "./pages/About";
import Pricing from "./pages/Pricing";
import Testimonials from "./pages/Testimonials";
import DataCollection from "./pages/DataCollection";
import DataAnalysis from "./pages/DataAnalysis";
import DataInterpretation from "./pages/DataInterpretation";
import Surveys from "./pages/Surveys";
import CaseStudies from "./pages/CaseStudies";
import Training from "./pages/Training";
import Translation from "./pages/Translation";
import ResearchAnalysis from "./pages/ResearchAnalysis";
import Tutoring from "./pages/Tutoring";

function Router() {
  return (
    <Switch>
      <Route path={"/"} component={Home} />
      <Route path={"/about"} component={About} />
      <Route path={"/pricing"} component={Pricing} />
      <Route path={"/testimonials"} component={Testimonials} />
      <Route path={"/data-collection"} component={DataCollection} />
      <Route path={"/data-analysis"} component={DataAnalysis} />
      <Route path={"/data-interpretation"} component={DataInterpretation} />
      <Route path={"/surveys"} component={Surveys} />
      <Route path={"/case-studies"} component={CaseStudies} />
      <Route path={"/training"} component={Training} />
      <Route path={"/translation"} component={Translation} />
      <Route path={"/research-analysis"} component={ResearchAnalysis} />
      <Route path={"/tutoring"} component={Tutoring} />
      <Route path={"/contact"} component={Contact} />
      <Route path={"/404"} component={NotFound} />
      <Route component={NotFound} />
    </Switch>
  );
}

function App() {
  return (
    <ErrorBoundary>
      <LanguageProvider>
        <ThemeProvider defaultTheme="light">
          <TooltipProvider>
            <Toaster />
            <Router />
            <FloatingWhatsApp />
          </TooltipProvider>
        </ThemeProvider>
      </LanguageProvider>
    </ErrorBoundary>
  );
}

export default App;

## 2. ملف LanguageContext.tsx
================================================================================

import React, { createContext, useContext, useState, useEffect } from 'react';
import { Language, getTranslation } from '@/translations';

interface LanguageContextType {
  language: Language;
  setLanguage: (lang: Language) => void;
  t: (path: string) => string;
  dir: 'ltr' | 'rtl';
}

const LanguageContext = createContext<LanguageContextType | undefined>(undefined);

export const LanguageProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [language, setLanguage] = useState<Language>('ar');

  useEffect(() => {
    const savedLanguage = localStorage.getItem('language') as Language | null;
    const browserLanguage = navigator.language.startsWith('ar') ? 'ar' : 'en';
    const initialLanguage = savedLanguage || browserLanguage;
    
    setLanguage(initialLanguage);
    document.documentElement.dir = initialLanguage === 'ar' ? 'rtl' : 'ltr';
    document.documentElement.lang = initialLanguage;
  }, []);

  const handleSetLanguage = (lang: Language) => {
    setLanguage(lang);
    localStorage.setItem('language', lang);
    document.documentElement.dir = lang === 'ar' ? 'rtl' : 'ltr';
    document.documentElement.lang = lang;
  };

  const t = (path: string): string => getTranslation(language, path);
  const dir = language === 'ar' ? 'rtl' : 'ltr';

  return (
    <LanguageContext.Provider value={{ language, setLanguage: handleSetLanguage, t, dir }}>
      {children}
    </LanguageContext.Provider>
  );
};

export const useLanguage = (): LanguageContextType => {
  const context = useContext(LanguageContext);
  if (!context) {
    throw new Error('useLanguage must be used within LanguageProvider');
  }
  return context;
};

import { MessageCircle } from "lucide-react";
import { useLanguage } from "@/contexts/LanguageContext";

export default function FloatingWhatsApp() {
  const { language } = useLanguage();
  const phoneNumber = "966546530264";
  const message = language === 'ar' 
    ? 'السلام عليكم، أود الاستفسار عن خدماتكم الإحصائية'
    : 'Hello, I would like to inquire about your statistical services';
  
  const whatsappUrl = `https://wa.me/${phoneNumber}?text=${encodeURIComponent(message)}`;

  return (
    <a
      href={whatsappUrl}
      target="_blank"
      rel="noopener noreferrer"
      className="fixed bottom-6 right-6 z-40 w-14 h-14 bg-green-500 hover:bg-green-600 text-white rounded-full flex items-center justify-center shadow-lg hover:shadow-xl transition transform hover:scale-110 animate-bounce"
      title={language === 'ar' ? 'تواصل عبر الواتساب' : 'Contact via WhatsApp'}
    >
      <MessageCircle className="w-7 h-7" />
    </a>
  );
}
