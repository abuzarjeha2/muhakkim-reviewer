async function callClaudeJSON(system, user) {
  const res = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "claude-sonnet-4-20250514",
      max_tokens: 1000,
      system: system + "\n\nأجب فقط بـ JSON صحيح بدون أي نص إضافي أو backticks",
      messages: [{ role: "user", content: user }],
    }),
  });
  if (!res.ok) throw new Error("API Error " + res.status);
  const d = await res.json();
  const txt = d.content.map(b => b.text || "").join("").replace(/```json|```/g, "").trim();
  return JSON.parse(txt);
}

function readAsText(file) {
  return new Promise((res, rej) => {
    const r = new FileReader();
    r.onload = () => res(r.result);
    r.onerror = () => rej(new Error("فشل القراءة"));
    r.readAsText(file, "utf-8");
  });
}

function ScoreBadge({ score }) {
  const [color, label] = score >= 85 ? ["#059669","ممتاز"] : score >= 70 ? ["#d97706","جيد جداً"] : score >= 55 ? ["#dc2626","مقبول"] : ["#7f1d1d","يحتاج تحسين"];
  return (
    <div style={{ display:"inline-flex", alignItems:"center", gap:6, background:color+"18", border:`1.5px solid ${color}40`, borderRadius:8, padding:"4px 12px", color, fontWeight:700, fontSize:14 }}>
      <span style={{ fontSize:18 }}>{score}%</span>
      <span style={{ fontSize:12, opacity:0.8 }}>{label}</span>
    </div>
  );
}

function AnalysisView() {
  const [file, setFile] = useState(null);
  const [title, setTitle] = useState("");
  const [degree, setDegree] = useState("doctorate");
  const [selected, setSelected] = useState(["researcher","internal","external"]);
  const [loading, setLoading] = useState(false);
  const [progress, setProgress] = useState({ step:"", percent:0 });
  const [results, setResults] = useState(null);
  const [error, setError] = useState("");
  const fileRef = useRef();

  const toggle = id => setSelected(p => p.includes(id) ? p.filter(r=>r!==id) : [...p,id]);

  const handleDrop = useCallback(e => { e.preventDefault(); const f=e.dataTransfer.files[0]; if(f) setFile(f); }, []);

  const analyze = async () => {
    if (!file && !title) { setError("يرجى رفع ملف أو إدخال عنوان الرسالة"); return; }
    if (selected.length === 0) { setError("يرجى اختيار فئة واحدة على الأقل"); return; }
    setError(""); setLoading(true); setResults(null);
    try {
      let content = "";
      if (file) {
        setProgress({ step:"جاري قراءة الملف...", percent:10 });
        if (file.type === "text/plain") {
          const txt = await readAsText(file);
          content = txt.length > 5000 ? txt.slice(0,5000)+"...[مقتصر]" : txt;
        } else {
          content = `[ملف: ${file.name} | ${(file.size/1024).toFixed(1)} كيلوبايت]`;
        }
      }
      const ctx = `عنوان الرسالة: ${title||"غير محدد"}\nالدرجة: ${degree==="doctorate"?"دكتوراه":"ماجستير"}\n${content}`;
      const reports = {};
      for (let i=0; i<selected.length; i++) {
        const role = selected[i];
        const ri = ROLES[role];
        setProgress({ step:`جاري إعداد تقرير ${ri.label}...`, percent: 20+Math.round(((i+1)/selected.length)*70) });
        const sys = `أنت ${ri.label} أكاديمي متخصص في تحكيم رسائل ${degree==="doctorate"?"الدكتوراه":"الماجستير"}.
أجب فقط بـ JSON بالشكل:
{"overallScore":<0-100>,"scores":{"structure":<0-100>,"methodology":<0-100>,"content":<0-100>,"language":<0-100>,"references":<0-100>},"summary":"<3-4 جمل>","observations":[{"severity":"critical|major|minor","page":"<رقم أو null>","section":"<اسم>","observation":"<الملاحظة>","explanation":"<تفسير>","solution":"<حل عملي>","example":"<مثال>"}],"recommendation":"<التوصية النهائية>"}`;
        try {
          reports[role] = await callClaudeJSON(sys, `قيّم هذه الرسالة من منظور ${ri.label}:\n${ctx}\nقدّم 4-6 ملاحظات متنوعة مع حلول عملية.`);
        } catch {
          reports[role] = { overallScore:0, summary:"تعذّر إنشاء التقرير. تأكد من الاتصال وأعد المحاولة.", observations:[], recommendation:"—" };
        }
      }
      setProgress({ step:"اكتمل التحليل!", percent:100 });
      setResults({ reports, generatedAt: new Date().toLocaleString("ar") });
    } catch(e) {
      setError("خطأ: " + e.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div style={{ padding:"24px 0" }}>
      <div style={{ background:"#fff", border:"1.5px solid #e5e7eb", borderRadius:16, padding:28, marginBottom:24, boxShadow:"0 2px 12px #0000000c" }}>
        <h3 style={{ margin:"0 0 20px", fontSize:18, fontWeight:700, color:"#1e3a5f" }}>📤 رفع الرسالة وضبط التحليل</h3>
        <div onDrop={handleDrop} onDragOver={e=>e.preventDefault()} onClick={()=>fileRef.current?.click()}
          style={{ border:`2px dashed ${file?"#059669":"#d1d5db"}`, borderRadius:12, padding:"32px 20px", textAlign:"center", cursor:"pointer", background:file?"#f0fdf4":"#f9fafb", marginBottom:20, transition:"all 0.2s" }}>
          <input ref={fileRef} type="file" style={{ display:"none" }} accept=".pdf,.doc,.docx,.txt" onChange={e=>setFile(e.target.files[0])} />
          {file ? (
            <div><span style={{ fontSize:36 }}>✅</span><p style={{ margin:"8px 0 0", fontSize:14, fontWeight:700, color:"#059669" }}>{file.name}</p><p style={{ margin:"4px 0 0", fontSize:12, color:"#6b7280" }}>{(file.size/1024).toFixed(1)} كيلوبايت</p></div>
          ) : (
            <div><span style={{ fontSize:48 }}>📄</span><p style={{ margin:"12px 0 4px", fontSize:15, fontWeight:600, color:"#374151" }}>اسحب الملف هنا أو انقر للاختيار</p><p style={{ margin:0, fontSize:12, color:"#9ca3af" }}>PDF · Word · TXT</p></div>
          )}
        </div>
        <div style={{ display:"grid", gridTemplateColumns:"1fr auto", gap:12, marginBottom:20 }}>
          <input value={title} onChange={e=>setTitle(e.target.value)} placeholder="عنوان الرسالة (اختياري)"
            style={{ padding:"12px 16px", borderRadius:10, border:"1.5px solid #d1d5db", fontSize:14, color:"#1f2937", outline:"none", fontFamily:"inherit" }} />
          <select value={degree} onChange={e=>setDegree(e.target.value)}
            style={{ padding:"12px 16px", borderRadius:10, border:"1.5px solid #d1d5db", fontSize:14, background:"#fff", cursor:"pointer", fontFamily:"inherit" }}>
            <option value="doctorate">دكتوراه</option>
            <option value="master">ماجستير</option>
          </select>
        </div>
        <div style={{ marginBottom:20 }}>
          <p style={{ margin:"0 0 10px", fontSize:13, fontWeight:600, color:"#374151" }}>اختر الفئات المطلوبة:</p>
          <div style={{ display:"flex", gap:10, flexWrap:"wrap" }}>
            {Object.values(ROLES).map(role => {
              const active = selected.includes(role.id);
              return (
                <button key={role.id} onClick={()=>toggle(role.id)}
                  style={{ padding:"10px 18px", borderRadius:10, border:`2px solid ${active?role.color:"#e5e7eb"}`, background:active?role.color+"15":"#fff", color:active?role.color:"#6b7280", fontWeight:active?700:500, fontSize:14, cursor:"pointer", display:"flex", alignItems:"center", gap:8, transition:"all 0.15s", fontFamily:"inherit" }}>
                  {role.icon} {role.label} {active&&"✓"}
                </button>
              );
            })}
          </div>
        </div>
        {error && <div style={{ marginBottom:16, padding:"10px 16px", background:"#fef2f2", border:"1px solid #fca5a5", borderRadius:8, color:"#dc2626", fontSize:14 }}>⚠️ {error}</div>}
        <button onClick={analyze} disabled={loading}
          style={{ width:"100%", padding:"14px 24px", background:loading?"#9ca3af":"linear-gradient(135deg,#1e3a5f,#2563eb)", color:"#fff", border:"none", borderRadius:12, fontSize:16, fontWeight:700, cursor:loading?"not-allowed":"pointer", fontFamily:"inherit", boxShadow:loading?"none":"0 4px 20px #2563eb40", transition:"all 0.2s" }}>
          {loading ? `⏳ ${progress.step}` : "🔍 بدء التحليل والتحكيم الذكي"}
        </button>
        {loading && (
          <div style={{ marginTop:12 }}>
            <div style={{ height:6, background:"#e5e7eb", borderRadius:3, overflow:"hidden" }}>
              <div style={{ height:"100%", width:`${progress.percent}%`, background:"linear-gradient(90deg,#2563eb,#7c3aed)", borderRadius:3, transition:"width 0.5s ease" }} />
            </div>
            <p style={{ textAlign:"center", fontSize:12, color:"#6b7280", margin:"6px 0 0" }}>{progress.percent}%</p>
          </div>
        )}
      </div>
      {results && (
        <div>
          <div style={{ display:"flex", alignItems:"center", justifyContent:"space-between", marginBottom:20, flexWrap:"wrap", gap:10 }}>
            <div>
