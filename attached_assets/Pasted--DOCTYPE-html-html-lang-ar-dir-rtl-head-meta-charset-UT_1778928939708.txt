function switchView(view) {
  document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));
  document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
  document.getElementById('view-' + view).classList.add('active');
  event.target.classList.add('active');
}

function setRole(role) {
  currentRole = role;
}

// ─────────────────────────────────────────
//  ANNOTATION
// ─────────────────────────────────────────
function selectPara(paraId, label) {
  if (selectedPara) {
    document.getElementById(selectedPara).classList.remove('selected');
  }
  selectedPara = paraId;
  document.getElementById(paraId).classList.add('selected');
  document.getElementById('comment-para-label').textContent = 'تعليق على: ' + label;
  const form = document.getElementById('comment-form');
  form.classList.add('open');
  document.getElementById(paraId).appendChild(form);
  document.getElementById('comment-text').focus();
}

function closeComment() {
  document.getElementById('comment-form').classList.remove('open');
  if (selectedPara) {
    document.getElementById(selectedPara).classList.remove('selected');
    selectedPara = null;
  }
}

function saveComment() {
  const text = document.getElementById('comment-text').value.trim();
  if (!text) return;
  const type = document.getElementById('comment-type').value;
  const paraLabel = document.getElementById('comment-para-label').textContent.replace('تعليق على: ', '');

  const comment = {
    id: Date.now(),
    para: selectedPara,
    paraLabel,
    text,
    type,
    time: 'الآن',
    role: currentRole
  };
  comments.push(comment);

  // Mark paragraph as having comment
  document.getElementById(selectedPara).classList.add('has-comment');
  const existing = document.getElementById(selectedPara).querySelector('.comment-bubble');
  if (existing) {
    const count = parseInt(existing.textContent.match(/\d+/)[0]) + 1;
    existing.textContent = `💬 ${count} تعليقات`;
  } else {
    const strong = document.getElementById(selectedPara).querySelector('strong');
    const bubble = document.createElement('span');
    bubble.className = 'comment-bubble';
    bubble.textContent = '💬 ١ تعليق';
    document.getElementById(selectedPara).insertBefore(bubble, strong);
  }

  document.getElementById('comment-text').value = '';
  closeComment();
  renderCommentsSidebar();
  updateStats();
}

function renderCommentsSidebar() {
  const container = document.getElementById('comments-sidebar');
  if (comments.length === 0) {
    container.innerHTML = `<div class="empty-state"><div class="empty-icon">💬</div>لا توجد تعليقات بعد.</div>`;
    return;
  }
  container.innerHTML = comments.slice().reverse().map(c => `
    <div class="comment-card">
      <div class="comment-para">${c.paraLabel}</div>
      <div class="comment-text">${c.text}</div>
      <div class="comment-time">${typeLabel(c.type)} · ${c.time}</div>
    </div>
  `).join('');
}

function typeLabel(t) {
  return { note:'📝 ملاحظة', question:'❓ سؤال', correction:'✏️ تصحيح', praise:'✅ إشادة' }[t] || t;
}

      </div>
      <div class="score-stars" id="stars-${c.key}">
        ${[1,2,3,4,5].map(i => `<span class="star" onclick="setScore('${c.key}',${i})" data-val="${i}">★</span>`).join('')}
      </div>
    </div>
  `).join('');
}

function setScore(key, val) {
  scores[key] = val;
  const container = document.getElementById('stars-' + key);
  container.querySelectorAll('.star').forEach(s => {
    s.classList.toggle('lit', parseInt(s.dataset.val) <= val);
  });
  updateStats();
}

function setDecision(btn, type) {
  document.querySelectorAll('.decision-btn').forEach(b => b.className = 'decision-btn');
  btn.classList.add(type);
}

}

function renderChapters() {
  const list = document.getElementById('chapter-list');
  list.innerHTML = chapters.map((ch, i) => `
    <div class="chapter-item" id="${ch.id}">
      <div class="chapter-header">
        <div class="chapter-name">${ch.name}</div>
        <div class="chapter-pct" id="pct-${ch.id}">${ch.pct}%</div>
      </div>
      <div class="progress-bar">
        <div class="progress-fill" id="bar-${ch.id}" style="width:${ch.pct}%"></div>
      </div>
      <div class="chapter-actions">
        <input type="range" min="0" max="100" value="${ch.pct}"
          style="flex:1; accent-color:var(--gold);"
          oninput="updateChapterPct('${ch.id}',this.value)">
        <button class="status-chip chip-${ch.status}" id="chip-${ch.id}" onclick="cycleStatus('${ch.id}')">
          ${statusLabel(ch.status)}
        </button>
      </div>
    </div>
  `).join('');
  updateOverall();
}

function updateChapterPct(id, val) {
  const ch = chapters.find(c => c.id === id);
  ch.pct = parseInt(val);
  document.getElementById('pct-' + id).textContent = val + '%';
  document.getElementById('bar-' + id).style.width = val + '%';
  updateOverall();
}

function cycleStatus(id) {
  const ch = chapters.find(c => c.id === id);
  const order = ['pending','review','approved'];
  ch.status = order[(order.indexOf(ch.status) + 1) % 3];
  const chip = document.getElementById('chip-' + id);
  chip.className = `status-chip chip-${ch.status}`;
  chip.textContent = statusLabel(ch.status);
  updateStats();
}

function statusLabel(s) {
  return { pending:'⏳ قيد الانتظار', review:'🔄 تحت المراجعة', approved:'✅ مقبول' }[s];
}

function updateOverall() {
  const avg = Math.round(chapters.reduce((a,c) => a + c.pct, 0) / chapters.length);
  document.getElementById('overall-pct').textContent = avg + '%';
  document.getElementById('overall-bar').style.width = avg + '%';
}

//  QUESTIONS
// ─────────────────────────────────────────
function renderQuestions() {
  const list = document.getElementById('questions-list');
  if (questions.length === 0) {
    list.innerHTML = `<div class="empty-state"><div class="empty-icon">❓</div>لا توجد أسئلة بعد. أضف أول سؤال.</div>`;
    return;
  }
  list.innerHTML = questions.map(q => `
    <div class="question-item ${q.done ? 'done' : ''}" id="q-${q.id}">
      <span class="q-drag">⠿</span>
      <div class="q-priority priority-${q.priority}">${{'high':'↑','med':'→','low':'↓'}[q.priority]}</div>
      <div class="q-content">
        <div class="q-text">${q.text}</div>
        <div class="q-meta">
          <span class="tag tag-blue">${q.chapter}</span>
          ${q.done ? '<span class="tag tag-green">✓ تمت المناقشة</span>' : ''}
        </div>
      </div>
      <button class="q-check ${q.done ? 'checked' : ''}" onclick="toggleQuestion(${q.id})">✓</button>
    </div>
  `).join('');
  updateStats();
}

function addQuestion() {
  const text = document.getElementById('new-q-text').value.trim();
  if (!text) return;
  questions.push({
    id: qIdCounter++,
    text,
    priority: document.getElementById('new-q-priority').value,
    chapter: document.getElementById('new-q-chapter').value,
    done: false
  });
  document.getElementById('new-q-text').value = '';
  renderQuestions();
}

function toggleQuestion(id) {
  const q = questions.find(q => q.id === id);
  q.done = !q.done;
  renderQuestions();
}

// ─────────────────────────────────────────
//  STATS
// ─────────────────────────────────────────
function updateStats() {
  document.getElementById('stat-comments').textContent = comments.length;
  document.getElementById('stat-questions').textContent = questions.length;
  document.getElementById('stat-approved').textContent = chapters.filter(c => c.status === 'approved').length;

  const vals = Object.values(scores);
  if (vals.length > 0) {
    const avg = (vals.reduce((a,b) => a+b, 0) / vals.length).toFixed(1);
    document.getElementById('stat-score').textContent = avg + '/5';
  }
}
