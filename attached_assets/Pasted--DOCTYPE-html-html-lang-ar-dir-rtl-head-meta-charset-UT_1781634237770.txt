        // Render all categories
        function renderCategories(filteredTools = null) {
            const container = document.getElementById('categories-container');
            container.innerHTML = '';

            categoriesData.forEach(category => {
                let toolsToShow = category.tools;
                
                if (filteredTools) {
                    toolsToShow = category.tools.filter(tool => 
                        filteredTools.some(ft => ft.id === tool.id)
                    );
                    if (toolsToShow.length === 0) return;
                }

                const section = document.createElement('details');
                section.className = `bg-white border border-zinc-200 rounded-3xl overflow-hidden modern-shadow ${filteredTools ? 'ring-2 ring-blue-200' : ''}`;
                section.open = true;

                const headerHTML = `
                    <summary class="category-header flex items-center justify-between px-6 py-5 bg-white hover:bg-zinc-50 transition-colors">
                        <div class="flex items-center gap-x-4">
                            <div class="w-11 h-11 rounded-2xl flex items-center justify-center bg-${category.color}-100 text-${category.color}-600">
                                <i class="fa-solid ${category.icon} text-2xl"></i>
                            </div>
                            <div>
                                <div class="flex items-center gap-x-3">
                                    <span class="font-semibold text-2xl tracking-tight">${currentLang === 'ar' ? category.title_ar : category.title_en}</span>
                                    <span class="px-3 py-0.5 text-xs font-bold rounded-full bg-zinc-100 text-zinc-600">${category.count} ${currentLang === 'ar' ? 'أداة' : 'tools'}</span>
                                </div>
                            </div>
                        </div>
                        <div class="flex items-center gap-x-2 text-zinc-400">
                            <i class="fa-solid fa-chevron-down transition-transform details-arrow"></i>
                        </div>
                    </summary>
                `;

                let toolsHTML = `<div class="px-6 pb-6 pt-2 grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">`;

                toolsToShow.forEach(tool => {
                    const badgeText = currentLang === 'ar' ? tool.badge : tool.badge_en;

        // Search functionality
        function setupSearch() {
            const searchInput = document.getElementById('search-input');
            
            searchInput.addEventListener('input', () => {
                const term = searchInput.value.toLowerCase().trim();
                
                if (!term) {
                    renderCategories();
                    return;
                }

                const filtered = [];
                
                categoriesData.forEach(cat => {
                    cat.tools.forEach(tool => {
                        const title = (currentLang === 'ar' ? tool.title_ar : tool.title_en).toLowerCase();
                        const desc = (currentLang === 'ar' ? tool.desc_ar : tool.desc_en).toLowerCase();
                        
                        if (title.includes(term) || desc.includes(term)) {
                            filtered.push(tool);
                        }
                    });
                });
                
                renderCategories(filtered.length > 0 ? filtered : null);
            });
        }

        // Language switcher
        function switchLanguage(lang) {
            currentLang = lang;
            
            // Update buttons
            document.getElementById('lang-ar').classList.toggle('active-lang', lang === 'ar');
            document.getElementById('lang-ar').classList.toggle('bg-white', lang === 'ar');
            document.getElementById('lang-ar').classList.toggle('shadow-sm', lang === 'ar');
            document.getElementById('lang-ar').classList.toggle('text-zinc-900', lang === 'ar');
            document.getElementById('lang-ar').classList.toggle('text-zinc-500', lang !== 'ar');

            document.getElementById('lang-en').classList.toggle('active-lang', lang === 'en');
            document.getElementById('lang-en').classList.toggle('bg-white', lang === 'en');
            document.getElementById('lang-en').classList.toggle('shadow-sm', lang === 'en');
            document.getElementById('lang-en').classList.toggle('text-zinc-900', lang === 'en');
            document.getElementById('lang-en').classList.toggle('text-zinc-500', lang !== 'en');

            // Update direction
            document.documentElement.lang = lang;
            document.documentElement.dir = lang === 'ar' ? 'rtl' : 'ltr';

            // Update search placeholder
            const search = document.getElementById('search-input');
            search.placeholder = translations[lang].search_placeholder;

            // Re-render everything
            renderCategories();
        }

                }
