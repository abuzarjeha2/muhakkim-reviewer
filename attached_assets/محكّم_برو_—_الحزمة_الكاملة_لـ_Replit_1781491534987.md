# 📦 محكّم برو — الحزمة الكاملة الجاهزة لـ Replit

> أنشئ مشروع **Node.js** جديد على Replit، ثم أنشئ الملفات التالية بنفس الأسماء وانسخ محتوى كل ملف.
> هيكل المشروع:
> ```
> muhakkim-pro/
> ├── package.json
> ├── server.js
> ├── provider.js
> ├── prompts.js
> ├── preprocess.js
> ├── router.js
> └── public/
>     └── index.html
> ```
> **مهم:** بعد الإنشاء، افتح تبويب **Secrets** (القفل) وأضف `AI_API_KEY` = مفتاح Gemini من Google AI Studio.

const PROVIDER_CONFIG = {
  endpoint: "https://generativelanguage.googleapis.com/v1beta/models",
  apiKeyEnv: "AI_API_KEY",
  models: {
    economy: "gemini-2.5-flash-lite",
    standard: "gemini-2.5-flash",
    premium: "gemini-2.5-pro", // غيّره لـ gemini-3.1-pro إن توفّر في حسابك
  },
};

const TIER_PROVIDER = { economy: "gemini", standard: "gemini", premium: "gemini" };

async function callGemini({ model, system, userContent, maxTokens }) {
  const apiKey = process.env[PROVIDER_CONFIG.apiKeyEnv];
  if (!apiKey) throw new Error("مفتاح API غير موجود في Replit Secrets (AI_API_KEY)");
  const url = `${PROVIDER_CONFIG.endpoint}/${model}:generateContent?key=${apiKey}`;
  const res = await fetch(url, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      systemInstruction: { parts: [{ text: system }] },
      contents: [{ role: "user", parts: [{ text: userContent }] }],
      generationConfig: { maxOutputTokens: maxTokens },
    }),
  });
  if (!res.ok) throw new Error(`فشل Gemini (${res.status}): ${await res.text()}`);
  const data = await res.json();
  return data.candidates?.[0]?.content?.parts?.map((p) => p.text).join("\n") || "";
}

async function callProvider({ tier, system, userContent, maxTokens = 4000 }) {
  const model = PROVIDER_CONFIG.models[tier];
  if (!model) throw new Error(`مستوى غير معروف: ${tier}`);
  const provider = TIER_PROVIDER[tier];
  if (provider === "gemini") return callGemini({ model, system, userContent, maxTokens });
  throw new Error(`مزوّد غير مدعوم: ${provider}`);
}

module.exports = { callProvider, PROVIDER_CONFIG, TIER_PROVIDER };

## 📄 الملف 4 — `preprocess.js` (المعالجة المحلية المجانية)

```javascript
const pdfParse = require("pdf-parse");
const mammoth = require("mammoth");

async function extractText(buffer, mimeType) {
  if (mimeType === "application/pdf") {
    const data = await pdfParse(buffer);
    return data.text;
  }
  if (mimeType === "application/vnd.openxmlformats-officedocument.wordprocessingml.document") {
    const result = await mammoth.extractRawText({ buffer });
    return result.value;
  }
  throw new Error("صيغة غير مدعومة. استخدم PDF أو DOCX");
}

function extractReferences(text) {
  const refSection = text.split(/(?:المراجع|References|REFERENCES|Bibliography)/i).pop() || "";
  const dois = [...refSection.matchAll(/10\.\d{4,9}\/[-._;()/:A-Z0-9]+/gi)].map((m) => m[0]);
  const entries = refSection
    .split(/\n(?=\[\d+\]|\d+\.\s|\d+\)\s)/)
    .map((s) => s.trim())
    .filter((s) => s.length > 20);
  return { count: entries.length, dois: [...new Set(dois)], entries: entries.slice(0, 100) };
}

async function verifyDOI(doi) {
  try {
    const res = await fetch(`https://api.crossref.org/works/${encodeURIComponent(doi)}`, {
      headers: { "User-Agent": "MuhakkimPro/1.0 (mailto:info@muhakkim.com)" },
    });
    if (!res.ok) return { doi, valid: false, reason: `HTTP ${res.status}` };
    const data = await res.json();
    const w = data.message;
    return {
      doi, valid: true,
      title: w.title?.[0] || "",
      year: w["published"]?.["date-parts"]?.[0]?.[0] || null,
      authors: (w.author || []).map((a) => a.family || "").filter(Boolean).join(", "),
      journal: w["container-title"]?.[0] || "",
    };
  } catch (e) {
    return { doi, valid: false, reason: e.message };
  }
}

async function verifyReferences(dois) {
  const results = [];
  for (const doi of dois.slice(0, 50)) {
    results.push(await verifyDOI(doi));
    await new Promise((r) => setTimeout(r, 120));
  }
  const valid = results.filter((r) => r.valid).length;
  return { total: results.length, valid, invalid: results.length - valid, results };
}

function ngrams(text, n = 5) {
  const words = text.toLowerCase().replace(/[^\u0600-\u06FF\w\s]/g, "").split(/\s+/).filter(Boolean);
  const grams = new Set();
  for (let i = 0; i <= words.length - n; i++) grams.add(words.slice(i, i + n).join(" "));
  return grams;
}

function similarity(textA, textB) {
  const a = ngrams(textA), b = ngrams(textB);
  if (a.size === 0 || b.size === 0) return 0;
  let inter = 0;
  for (const g of a) if (b.has(g)) inter++;
  return inter / Math.min(a.size, b.size);
}

function checkPlagiarismLocal(text, corpus) {
  const matches = (corpus || [])
    .map((doc) => ({ id: doc.id, title: doc.title, score: similarity(text, doc.text) }))
    .filter((m) => m.score > 0.05)
    .sort((a, b) => b.score - a.score)
    .slice(0, 10);
  return { topMatches: matches, highestScore: matches[0]?.score || 0 };
}

module.exports = {
  extractText, extractReferences, verifyReferences, verifyDOI,
  checkPlagiarismLocal, similarity,
};
```

## 📄 الملف 5 — `router.js` (تقدير التكلفة)

```javascript
const { SERVICES } = require("./prompts");

const TIER_COST = {
  economy: { in: 0.10, out: 0.40 },
  standard: { in: 0.15, out: 0.60 },
  premium: { in: 1.25, out: 5.00 },
};

function estimateFullReview(serviceIds, inTokens = 8000, outTokens = 1500) {
  let total = 0;
  const breakdown = serviceIds.map((id) => {
    const svc = SERVICES[id];
    if (!svc) return { service: id, error: "غير معروفة" };
    const c = TIER_COST[svc.tier];
    const cost = (inTokens / 1e6) * c.in + (outTokens / 1e6) * c.out;
    total += cost;
    return { service: id, tier: svc.tier, estimatedUSD: +cost.toFixed(4) };
  });
  return { breakdown, totalUSD: +total.toFixed(4) };
}

module.exports = { estimateFullReview, TIER_COST };
```

const cors = require("cors");

const app = express();
app.use(cors());

app.use(express.raw({
  type: ["application/pdf", "application/vnd.openxmlformats-officedocument.wordprocessingml.document"],
  limit: "25mb",
}));
app.use(express.static(path.join(__dirname, "public"))); // يقدّم index.html

// تخزين مؤقت + طابور + إعادة محاولة
const cache = new Map();
function cacheKey(s, t) { let h = 0; for (let i = 0; i < t.length; i++) h = (h * 31 + t.charCodeAt(i)) | 0; return `${s}:${h}`; }
let active = 0; const MAX = 3; const queue = [];
async function acquire() { if (active < MAX) { active++; return; } await new Promise((r) => queue.push(r)); active++; }
function release() { active--; const n = queue.shift(); if (n) n(); }
async function withRetry(fn, retries = 3) {
  for (let i = 1; i <= retries; i++) {
    try { return await fn(); }
    catch (e) { if (i === retries) throw e; await new Promise((r) => setTimeout(r, 1000 * 2 ** (i - 1))); }
  }
}
async function runService(service, text) {
  const cfg = SERVICES[service];
  if (!cfg) throw new Error(`خدمة غير معروفة: ${service}`);
  const key = cacheKey(service, text);
  if (cache.has(key)) return { result: cache.get(key), cached: true, tier: cfg.tier, label: cfg.label };
  await acquire();
  try {
    const result = await withRetry(() => callProvider({ tier: cfg.tier, system: cfg.system, userContent: text }));
    cache.set(key, result);
    return { result, cached: false, tier: cfg.tier, label: cfg.label };
  } finally { release(); }
}

app.get("/api/health", (req, res) => res.json({ status: "محكّم برو يعمل ✓", services: Object.keys(SERVICES).length }));

app.get("/api/services", (req, res) =>
  res.json({ services: Object.entries(SERVICES).map(([id, v]) => ({ id, label: v.label, tier: v.tier })) }));

app.post("/api/estimate", (req, res) => {
  const { services } = req.body;
  if (!Array.isArray(services)) return res.status(400).json({ error: "مطلوب: services[]" });
  res.json(estimateFullReview(services));
});

app.post("/api/extract-file", async (req, res) => {
  try {
    const text = await pre.extractText(req.body, req.headers["content-type"]);
    res.json({ textLength: text.length, text, references: pre.extractReferences(text) });
  } catch (e) { res.status(400).json({ error: e.message }); }
});

app.post("/api/verify-references", async (req, res) => {
  const { dois, text } = req.body;
  try {
    const list = dois || pre.extractReferences(text || "").dois;
    res.json(await pre.verifyReferences(list));
  } catch (e) { res.status(500).json({ error: e.message }); }
});

app.post("/api/plagiarism-local", (req, res) => {
  const { text, corpus } = req.body;
  if (!text) return res.status(400).json({ error: "مطلوب: text" });
  res.json(pre.checkPlagiarismLocal(text, corpus || []));
});

app.post("/api/analyze", async (req, res) => {
  const { service, text } = req.body;
  if (!service || !text) return res.status(400).json({ error: "مطلوب: service و text" });
  try { res.json({ service, ...(await runService(service, text)) }); }
  catch (e) { res.status(500).json({ error: e.message }); }
});

app.post("/api/analyze-batch", async (req, res) => {
  const { services, text } = req.body;
  if (!Array.isArray(services) || !text) return res.status(400).json({ error: "مطلوب: services[] و text" });
  const results = {};
  await Promise.all(services.map(async (s) => {
    try { results[s] = await runService(s, text); }
    catch (e) { results[s] = { error: e.message }; }
  }));
  res.json({ results });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`✓ محكّم برو يعمل على المنفذ ${PORT}`));
